/* ==========================================

function render(){
    const container =
        document.getElementById("products");
    container.innerHTML = "";
    const categorias = [
        "Quesos",
        "Quesillos",
        "Dulces",
        "Salames"
    ];
    categorias.forEach(categoria => {
        const productosCategoria =
            productos
            .map((p,i)=>({p,i}))
            .filter(x=>x.p.categoria === categoria);
        if(productosCategoria.length === 0){
            return;
        }
        let html = `
        <div class="categoria">
            <div class="titulo-categoria">
                ${categoria}
            </div>
        `;
        productosCategoria.forEach(({p,i}) => {
            html += `
            <div class="product">
                <div class="left">
                    <img
                        src="${p.img}"
                        class="foto"
                        onerror="this.style.display='none'"
                    >
                    <div class="info-prod">
                        <div class="name">
                            ${p.nombre}
                        </div>
                        <div class="price">
                            $${p.precio.toLocaleString('es-AR')}
                        </div>
                    </div>
                </div>
                <div class="controls">
                    <button
                        class="btn-qty"
                        onclick="cambiar(${i},-1)"
                    >
                        -

                    <div class="cantidad">
                        ${carrito[i]}

                </div>
            </div>
            `;
        });
        html += `</div>`;
        container.innerHTML += html;
    });
}
/* ==========================================
   CAMBIAR CANTIDAD
========================================== */
function cambiar(i,q){
    carrito[i] += q;
    if(carrito[i] < 0){
        carrito[i] = 0;
    }
    actualizarTotal();
    render();
}
/* ==========================================
   TOTAL
========================================== */
function actualizarTotal(){
    let t = 0;
    productos.forEach((p,i)=>{
        t += p.precio * carrito[i];
    });
    document.getElementById("total").innerText =
        t.toLocaleString('es-AR');
}
/* ==========================================
   WHATSAPP
========================================== */
function enviarPedido(){
    if(!carrito.some(c=>c>0)){
        return alert("Agrega al menos un producto");
    }
    let nombre =
        document.getElementById("nombre").value ||
        "Cliente";
    let m =
        `🧀 *PEDIDO - LA FÁBRICA*\n` +
        `👤 Cliente: ${nombre}\n` +
        `--------------------------\n`;
    const categorias = [
        "Quesos",
        "Quesillos",
        "Dulces",
        "Salames"
    ];
    categorias.forEach(categoria=>{
        const seleccionados =
            productos
            .map((p,i)=>({p,i}))
            .filter(x =>
                x.p.categoria === categoria &&
                carrito[x.i] > 0
            );
        if(seleccionados.length === 0){
            return;
        }
        m += `\n*${categoria.toUpperCase()}*\n`;
        seleccionados.forEach(({p,i})=>{
            m +=
                `• ${p.nombre} x${carrito[i]} ` +
                `($${(p.precio*carrito[i]).toLocaleString('es-AR')})\n`;
        });
    });
    m +=
        `\n💰 *Total: $${document.getElementById("total").innerText}*`;
    let d =
        document.getElementById("detalle").value;
    if(d){
        m += `\n\n📝 Nota: ${d}`;
    }
    window.open(
        "https://wa.me/5493816062471?text=" +
        encodeURIComponent(m)
    );
}

   IMPRIMIR
========================================== */
function imprimirPedido(){

    let nombre =
