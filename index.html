const express = require('express'); const cors = require('cors'); const bodyParser = require('body-parser'); const pdf = require('html-pdf-node'); const printer = require('pdf-to-printer'); const fs = require('fs'); const path = require('path'); const app = express(); // Habilitar CORS app.use(cors()); // Parsear todos los formatos posibles app.use(bodyParser.json({ limit: '10mb' })); app.use(bodyParser.text({ type: '*/*', limit: '10mb' })); app.use(bodyParser.urlencoded({ extended: true, limit: '10mb' })); const NOMBRE_IMPRESORA = "POS-80"; app.post('/imprimir', async (req, res) => { try { let htmlContent = ""; // Detectar formato de la petición if (typeof req.body === 'string') { try { const parsed = JSON.parse(req.body); htmlContent = parsed.html; } catch (e) { htmlContent = req.body; } } else if (req.body && req.body.html) { htmlContent = req.body.html; } if (!htmlContent) { return res.status(400).json({ status: 'error', message: 'No se recibió contenido HTML' }); } // Opciones optimizadas para impresión térmica oscura const options = { width: '80mm', printBackground: true, preferCSSPageSize: true }; const file = { content: htmlContent }; // Generar archivo PDF del ticket const pdfBuffer = await pdf.generatePdf(file, options); const filePath = path.join(__dirname, 'ticket.pdf'); fs.writeFileSync(filePath, pdfBuffer); // Imprimir en la POS-80 await printer.print(filePath, { printer: NOMBRE_IMPRESORA }); console.log("✅ Ticket impreso con éxito"); res.json({ status: 'ok', message: 'Ticket impreso correctamente' }); } catch (error) { console.error("❌ Error al imprimir:", error); res.status(500).json({ status: 'error', message: error.toString() }); } }); app.listen(3000, '0.0.0.0', () => { console.log('----------------------------------------------------'); console.log('Servidor de La Fábrica activo en el puerto 3000'); console.log('Impresora configurada: POS-80'); console.log('----------------------------------------------------'); });
